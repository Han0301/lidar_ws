cmake_minimum_required(VERSION 3.16)
project(lidar_mapping)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(octomap REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(occupancy_core src/occupancy_mapper.cpp)
target_include_directories(occupancy_core PUBLIC include ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(occupancy_core PUBLIC ${OCTOMAP_LIBRARIES})
add_executable(mapping_node src/mapping_node.cpp)
target_link_libraries(mapping_node occupancy_core Eigen3::Eigen)
ament_target_dependencies(mapping_node rclcpp sensor_msgs nav_msgs diagnostic_msgs std_srvs tf2_ros)
install(TARGETS mapping_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_occupancy test/test_occupancy.cpp)
  target_link_libraries(test_occupancy occupancy_core)
endif()
ament_package()
